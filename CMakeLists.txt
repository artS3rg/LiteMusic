cmake_minimum_required(VERSION 3.20)
project(LiteMusic VERSION 0.1.1 LANGUAGES C CXX)

if(NOT WIN32)
    if(APPLE)
        enable_language(OBJCXX)
    endif()
    include(cmake/Portable.cmake)
    return()
endif()

enable_language(RC)
add_executable(LiteMusic WIN32 src/main.cpp src/music.cpp src/music_win.cpp src/settings.cpp third_party/cjson/cJSON.c resources/LiteMusic.rc)
set_property(SOURCE resources/LiteMusic.rc APPEND PROPERTY OBJECT_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/resources/LiteMusic.ico;${CMAKE_CURRENT_SOURCE_DIR}/resources/LiteMusic.manifest;${CMAKE_CURRENT_SOURCE_DIR}/LICENSE;${CMAKE_CURRENT_SOURCE_DIR}/THIRD_PARTY_NOTICES")
target_compile_features(LiteMusic PRIVATE cxx_std_20)
target_compile_definitions(LiteMusic PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX WINVER=0x0A00 _WIN32_WINNT=0x0A00)
target_include_directories(LiteMusic PRIVATE third_party/cjson resources)
target_link_libraries(LiteMusic PRIVATE winhttp bcrypt crypt32 advapi32 mfplay ole32 uuid shell32 comctl32 gdi32 dwmapi runtimeobject)
if(MSVC)
    target_compile_options(LiteMusic PRIVATE /utf-8)
    target_link_options(LiteMusic PRIVATE /MANIFEST:NO)
    set_property(TARGET LiteMusic PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
elseif(MINGW)
    target_link_options(LiteMusic PRIVATE -municode -static -static-libstdc++ -static-libgcc)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        # Some MinGW distributions add a manifest in the startup objects.
        # Keep their normal link settings and embed only the app's own manifest.
        execute_process(COMMAND "${CMAKE_CXX_COMPILER}" -dumpspecs OUTPUT_VARIABLE gcc_specs)
        string(REGEX MATCH "\\*endfile:[^\n]*\n([^\n]*)" endfile_spec "${gcc_specs}")
        set(endfile_rule "${CMAKE_MATCH_1}")
        set(default_manifest "%{!shared:%:if-exists(default-manifest.o%s)}")
        string(FIND "${endfile_rule}" "${default_manifest}" has_default_manifest)
        if(NOT has_default_manifest EQUAL -1)
            string(REPLACE "${default_manifest}" "" endfile_rule "${endfile_rule}")
            set(manifest_specs "${CMAKE_CURRENT_BINARY_DIR}/litemusic-link.specs")
            file(WRITE "${manifest_specs}" "*endfile:\n${endfile_rule}\n\n")
            target_link_options(LiteMusic PRIVATE "-specs=${manifest_specs}")
        endif()
    endif()
endif()
